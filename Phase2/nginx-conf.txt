
upstream auth_cluster {
    server localhost:8080;
}

upstream order_cluster {
    server localhost:8082;
}

upstream inventory_cluster {
    server localhost:8081;
}

server {
    listen 8000; # The unified entryway for everything
    server_name localhost;

    # Directs any request starting with /auth/ to the auth_cluster
    location /auth/ {
        proxy_pass http://auth_cluster;
        proxy_set_header Host $host;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    }

    # Directs any request starting with /orders/ to the order_cluster
    location /orders/ {
        proxy_pass http://order_cluster;
        proxy_set_header Host $host;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    }

    # Directs any request starting with /inventory/ to the inventory_cluster
    location /inventory/ {
        proxy_pass http://inventory_cluster;
        proxy_set_header Host $host;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    }
}
